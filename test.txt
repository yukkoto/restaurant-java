package com.safezone.infrastructure.data;

import java.sql.SQLException;
import java.sql.PreparedStatement;
import java.sql.DriverManager;
import java.sql.Connection;

public class example {
 private static final String url = "jdbc:postgresql://localhost:5432/safe_zone";
 private static final String user = "acidicacidity";
 private static final String pswd = "0405";

 public void createTable() throws SQLException {
  try (Connection conn = DriverManager.getConnection(url, user, pswd)) {
   String sql = "CREATE TABLE IF NOT EXISTS cell123 (id SERIAL)";
   PreparedStatement ps = conn.prepareStatement(sql);
   ps.executeUpdate();
  } catch (SQLException e) {
      System.out.println(e);
  }
 }
}
